#include <stdio.h>

#define MAX 100

int main()
{
    int graph[MAX][MAX];
    int visited[MAX] = {0};
    int queue[MAX];
    int front = 0, rear = 0;
    int n, i, j, start, vertex;

    printf("Enter the number of vertices: ");
    scanf("%d", &n);

    printf("Enter the adjacency matrix:\n");
    for (i = 0; i < n; i++)
    {
        for (j = 0; j < n; j++)
            scanf("%d", &graph[i][j]);
    }

    printf("Enter the starting vertex (1 to %d): ", n);
    scanf("%d", &start);

    start--;

    queue[rear++] = start;
    visited[start] = 1;

    printf("BFS Traversal: ");

    while (front < rear)
    {
        vertex = queue[front++];

        printf("%d ", vertex + 1);

        for (i = 0; i < n; i++)
        {
            if (graph[vertex][i] == 1 && visited[i] == 0)
            {
                queue[rear++] = i;
                visited[i] = 1;
            }
        }
    }

    return 0;
}
