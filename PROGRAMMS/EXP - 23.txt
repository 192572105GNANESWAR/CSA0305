#include <stdio.h>
#define MAX 100
#define INF 99999
void dijkstra(int graph[MAX][MAX], int n, int source)
{
    int distance[MAX], visited[MAX];
    int i, j, min, next;
    for (i = 0; i < n; i++)
    {
        distance[i] = INF;
        visited[i] = 0;
    }
    distance[source] = 0;
    for (i = 0; i < n - 1; i++)
    {
        min = INF;
        next = -1;
        for (j = 0; j < n; j++)
        {
            if (!visited[j] && distance[j] < min)
            {
                min = distance[j];
                next = j;
            }
        }
        if (next == -1)
            break;
        visited[next] = 1;
        for (j = 0; j < n; j++)
        {
            if (!visited[j] && graph[next][j] != 0 &&
                distance[next] + graph[next][j] < distance[j])
            {
                distance[j] = distance[next] + graph[next][j];
            }
        }
    }
    printf("\nShortest distances from vertex %d:\n", source + 1);
    for (i = 0; i < n; i++)
    {
        if (distance[i] == INF)
            printf("Vertex %d : Not reachable\n", i + 1);
        else
            printf("Vertex %d : %d\n", i + 1, distance[i]);
    }
}
int main()
{
    int graph[MAX][MAX];
    int n, source;
    int i, j;
    printf("Enter the number of vertices: ");
    scanf("%d", &n);
    printf("Enter the weighted adjacency matrix:\n");
    printf("(Enter 0 if there is no edge)\n");
    for (i = 0; i < n; i++)
    {
        for (j = 0; j < n; j++)
        {
            scanf("%d", &graph[i][j]);
        }
    }
    printf("Enter the source vertex (1 to %d): ", n);
    scanf("%d", &source);
    dijkstra(graph, n, source - 1);
    return 0;
}
